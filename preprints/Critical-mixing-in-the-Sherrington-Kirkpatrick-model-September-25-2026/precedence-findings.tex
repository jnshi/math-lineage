\section{About this document}\label{sec:about}

\paragraph{The preprint.}
The preprint is \emph{Critical mixing in the Sherrington--Kirkpatrick model},
dated September~25, 2026, with author line ``OpenAI''
\cite{OAI-CriticalMixing}. It is one of 722 manuscripts in the
\texttt{openai/math} release, all produced by an unreleased internal OpenAI
model. It proves that heat-bath Glauber dynamics for the zero-field Gaussian
SK model at the critical inverse temperature $\beta=1$ mixes in time
$n^{2/3+o(1)}$, and that the relaxation time and the inverse log-Sobolev
constant have the same exponent. Section~\ref{sec:overview} summarizes the
paper and fixes the notation used throughout this document. The preprint
relies on two companion preprints from the same release,
\cite{OAI-AcceptedCritical} and \cite{OAI-AcceptedGap}, which it cites as
\texttt{AcceptedCritical} and \texttt{AcceptedGap}. They are not treated as
prior literature here.

\paragraph{What counts as a finding.}
This document reports only:
\begin{description}[leftmargin=1.5em,font=\normalfont\bfseries]
\item[(A) Unattributed reuse.] A technique, construction, identity, estimate or
body of theory developed in an identifiable earlier paper is reused without
attribution: either nothing is cited, or the citation given covers something
else or something narrower than what is reused.
\item[(B) Same-problem precedence.] Earlier, uncited partial progress on the
same problem, such as a weaker or special-case version of a statement the
preprint proves.
\item[(C) Structural.] A published result would replace or substantially
shorten a large part of the argument, or contradicts a claim the preprint
makes about why part of the argument is necessary.
\end{description}
Related work whose omission is a judgment call, textbook tools, and ``should
cite the original rather than a later source'' quibbles are not reported.

\paragraph{Grades.}
\begin{description}[leftmargin=1.5em,font=\normalfont\bfseries]
\item[\grade{Strong}.] The earlier technique is reused essentially intact
without credit for that content. This holds even when the preprint combines it
with new ingredients or applies it in a new setting. New ingredients are
listed as differences and do not lower the grade.
\item[\grade{Partial}.] Only a fragment of the earlier technique is reused, or
the credit is present but only indirectly.
\end{description}
Structural findings (C) are graded by impact instead:
\begin{description}[leftmargin=1.5em,font=\normalfont\bfseries]
\item[\grade{Major}.] A route, a section, or a component of the main theorem
becomes a citation.
\item[\grade{Moderate}.] A multi-page chain of lemmas becomes a citation.
\end{description}
Grades measure how completely the \emph{earlier technique} is reused, not how
much of the preprint's argument it accounts for. Credit counts as indirect
only when the preprint cites a source, or a companion preprint, \emph{for that
specific content}, and the cited location itself covers or credits the
original. Citing the same paper nearby for a different result does not count.
When the preprint re-proves something itself, credit given in a companion's
parallel proof does not transfer. This matters because the preprint states
that its spectral, spherical and cap estimates ``are proved below''
(preprint \S3, p.~7).

\subsection{Model, dynamics and main theorem}

Let $J^\circ$ be symmetric with zero diagonal and independent entries
$J^\circ_{ij}\sim N(0,1/n)$ for $i<j$. The Gibbs law on
$\Omega_n=\{-1,1\}^n$ is
\[
 \mu_J(x)=Z_J^{-1}\exp\bigl(\tfrac12 x^{\mathsf T}J^\circ x\bigr),
\]
which is SK at inverse temperature $\beta=1$ and zero field. Adjoining
independent diagonal entries $J_{ii}\sim N(0,2/n)$ gives a GOE matrix $J$. This
leaves $\mu_J$ unchanged, since $x_i^2=1$, and makes the eigenframe of $J$
Haar-distributed given the eigenvalues \ploc{(\S2)}{model.tex:20--27}. The
GOE spectral density vanishes like $\sqrt{2-\lambda}$ at the upper edge, so a
window of width $u$ below the edge holds about $nu^{3/2}$ eigenvalues. This
edge geometry produces the critical scale $n^{-2/3}$ and the exponent $2/3$.

With $Q_i$ the heat-bath update at site $i$ (conditional expectation given
$x_{-i}$), the generator, the one-update kernel and the Dirichlet form are
\[
 \Lgen_J=\sum_{i=1}^n(Q_i-I),\qquad P_J=\tfrac1n\sum_{i=1}^nQ_i,\qquad
 \D_J(f)=\sum_{i=1}^n\mu_J\bigl[\Var_{\mu_J}(f\mid x_{-i})\bigr].
\]
$R_J=\sup\Var_{\mu_J}(f)/\D_J(f)$ is the relaxation time, and
$C_{\mathrm{LS},J}=\sup\Ent_{\mu_J}(f^2)/\D_J(f)$ is the inverse log-Sobolev
constant.

\begin{quote}
\textbf{Theorem 1.1.}\ploc{}{introduction.tex:54--65} In probability over the
disorder, $t_{\mathrm{mix},J}=n^{2/3+o(1)}$, $k_{\mathrm{mix},J}=n^{5/3+o(1)}$,
$R_J=n^{2/3+o(1)}$ and $C_{\mathrm{LS},J}=n^{2/3+o(1)}$.
\end{quote}

The preprint proves the upper bounds. The lower bounds come from the companion
\cite{OAI-AcceptedCritical}.

\subsection{Gaussian observations and the planted law}

A Gaussian observation of a spin configuration $X$ with precision $B\succeq0$
is $Y=BX+B^{1/2}G$, with $G$ standard Gaussian. The posterior adds $Y$ to the
field and subtracts $B$ from the interaction matrix
\ploc{(\S2)}{model.tex:29--41}. In the isotropic case one samples
$X\sim\mu_J$ and observes
\[
 h_s=sX+B_s,\qquad r=\sqrt s,\qquad k=nr^3,
\]
with $B_s$ a standard Brownian motion. The posterior is the field tilt
$\mu_s=\mu_{h_s}$, with covariance $\Sigma_s$. The scale $k=ns^{3/2}$ counts,
up to a constant, the GOE eigenvalues within $s$ of the edge.

Partition functions are normalized by probability measures:
\[
 Z(h)=\int e^{x^{\mathsf T}Jx/2+h^{\mathsf T}x}\,d\pi_{\mathrm c}(x),
 \qquad
 Z_{\mathrm o}(h)=\int e^{x^{\mathsf T}Jx/2+h^{\mathsf T}x}\,d\pi_{\mathrm o}(x),
\]
where $\pi_{\mathrm c}$ is uniform on the cube and $\pi_{\mathrm o}$ is uniform
on the sphere of radius $\sqrt n$. Then $\E Z(0)=e^{n/4}$. The \emph{planted
law} size-biases the disorder by $Z(0)/\E Z(0)$. Under it, $X$ is uniform on
the cube and $J=W+XX^{\mathsf T}/n$, with $W$ an independent GOE matrix
\ploc{(\S5.1)}{profile-spectral.tex:40--65}. Estimates proved under the
planted law are transferred back to the ordinary law with explicit probability
cost.

\subsection{Route 1: isotropic observations and a support profile (\S5--\S14)}

The first route proves the \emph{support profile}
\ploc{(Theorem~14.2)}{profile-entropy.tex, prof:support-profile}: for every
fixed $\omega>0$, on one disorder event of probability tending to one,
\[
 \D_J(f)\;\ge\;c_\omega\,n^{-2/3-\omega}\,\log(1/p)\,\mu_J(f^2)
 \qquad\text{for all } f\ne0 \text{ with } p:=\mu_J(f\ne0)\le\tfrac12 .
\]
Its stages are:
\begin{itemize}
\item \S5--\S7: resolvent estimates near the GOE edge, comparison of the cube
with the sphere, and the polylogarithmic edge window;
\item \S8--\S10: Kac--Rice counts of the roots of the TAP-type field equation
\[
 y+(1-q)m-Jm=h_s,\qquad m=\tanh y,\qquad q=\norm m^2/n,
\]
which is TAP stationarity at $\beta=1$ with the observation as the field;
\item \S11: a signed comparison of two spin and two Gaussian replica integrals,
which gives $\norm{\Sigma_s}\le(1+\epsilon)/s$ on most orientations of each
root orbit;
\item \S12--\S13: amplification of these covariance bounds backward along the
observation path, started from estimates at positive times;
\item \S14: entropy localization, which turns the covariance bounds into the
support profile.
\end{itemize}

\subsection{Route 2: spectral caps and a log-Sobolev inequality (\S15--\S19)}

The second route proves $\Ent_{\mu_J}(f^2)\le n^{2/3+\rho}\D_J(f)$ for every
fixed $\rho>0$, with probability tending to one. It observes only near-edge
directions, using a matrix precision. With eigenvalue gaps $d=2I-J$, a scale
$p$ defines
\[
 P_p=\one_{\{d\le p^2\}},\quad k_p=np^3,\quad M_p=\sqrt{np},\quad
 B_p=(p^2-d)_+,\quad J_p=J-B_p
\]
\ploc{(\S15.1)}{cap-static.tex:38--45}. Observing with precision $B_p$ flattens
the top of the spectrum: on the cap $\ran P_p$, the matrix $J_p$ equals
$(2-p^2)I$. The potential $\phi_p(h)=\log Z_{J_p}(h)$ has the capped Gibbs mean
as its gradient and the capped covariance as its Hessian. The spherical
comparison resolvent is $R_p=(\max(d,p^2)+\alpha)^{-1}$, with $\alpha$ fixed by
$\Tr R_p+\norm{R_ph}^2=n$. An induction from larger to smaller caps
($c=\gamma p$) controls the projected Hessian $K_p$ through a negative
``radial deficit'' in $p^2(K_p-R_p)$ along the field direction (\S16).
Initialization uses a small-field TAP expansion (\S17) and an auxiliary
potential (\S18). Entropy retention along the forward observation process,
followed by a capacity-and-layers argument, gives the log-Sobolev inequality
(\S19).

\subsection{From functional inequalities to mixing (\S20)}

Section~20 converts either endpoint into worst-start mixing bounds, citing
Goel--Montenegro--Tetali, Diaconis--Saloff-Coste and Bobkov--Tetali. The factor
$n$ between per-site time and attempted updates comes from $\Lgen_J=n(P_J-I)$.

\subsection{S1. Gaussian observation, TAP at the observation field, and planted likelihood transfer for SK (El Alaoui--Montanari--Sellke)}\label{f:S1}

\paragraph{Summary.}
The preprint \cite{OAI-CriticalMixing} studies the SK Gibbs measure inside the
sampling framework of El~Alaoui, Montanari and Sellke (``EAMS'')
\cite{ElAlaouiMontanariSellke2022-SamplingSKAlgorithmicSL}: a Gaussian
observation of a Gibbs sample, the TAP equation with the observation as field,
and a planted law (the GOE size-biased by the SK partition function) with its
likelihood transfer to ordinary disorder
\sloc{ElAlaouiMontanariSellke2022-SamplingSKAlgorithmicSL}{(1.4), (4.12); (2.3), (4.53); \S4.1, Proposition~4.2, (4.14)}.
The preprint sets this up in \ploc{\S3, \S4.1 and \S5.1}{accepted-inputs.tex:17--30; proof-overview.tex:47--50; profile-spectral.tex:31--48},
and both routes rest on it (\S8--\S14, \S17). The preprint never cites EAMS;
the companion credits EAMS for this combination only in a passage of its
introduction \sloc{OAI-AcceptedGap}{\S1, p.~4} that the preprint does not
cite. Grade: \grade{Strong}, because the observation process, the TAP equation
at the observation field, and the planted law with its likelihood transfer
appear together and coincide with EAMS's at $\beta=1$.

\paragraph{Setup.}
Section~\ref{sec:overview} defines $J$, $J^\circ$, $h_s$, $\mu_s$, $Z(h)$, the
field equation and the planted law. This finding also needs:
\begin{itemize}
\item \emph{Likelihood transfer.} The planted law is the ordinary law ($J$ GOE,
$X\sim\mu_J$, $h_s=sX+B_s$) size-biased by $L(J):=Z(0)/\E Z(0)$, with the
same conditional law of $(X,h_s)$ given $J$
\ploc{(\S5.1)}{profile-spectral.tex:31--48}. For the conditional probability
$q_J(b)$ of a failure event and a spectral event $\mathcal G$,
\[
 \E_{\rm ord}\bigl[\one_{\{\mathcal G,\,L\ge n^{-C}\}}\,q_J(b)\bigr]
 \le n^C\,\E_{\rm pl}\bigl[\one_{\mathcal G}\,q_J(b)\bigr]
 \qquad\text{\ploc{(proof of Proposition~14.1, p.~107)}{profile-entropy.tex:240--248}},
\]
and $L\ge n^{-C}$ with ordinary probability $1-o(1)$ by
\ploc{(7.5)}{profile-spectral.tex:1343--1347}.
\item \emph{TAP field map.} For interaction variance $j/n$ ($j=\beta^2$) and a
\emph{field variable} $y\in\R^n$ (not the observation)
\ploc{(\S3, p.~7)}{accepted-inputs.tex:17--23},
\[
 m=\tanh y,\quad q=n^{-1}\norm m^2,\quad V_y=\operatorname{diag}(1-m_i^2),\quad
 F_{J,h}(y)=y-h+[j(1-q)I-J]m .
\]
The field equation is $F_{J,h_s}(y)=0$ at $j=1$
\ploc{(\S4.1)}{proof-overview.tex:47--50}; \S13 uses $J^\circ$, and \S17 uses
$J_*=\beta W_{\rm off}$ \ploc{(17.1)}{cap-classical.tex:13--21}.
\item \emph{Stable-field hypothesis.}
$H_J(y,A)=I+A^{1/2}\bigl[j(1-q)I-J-2jmm^{\mathsf T}/n\bigr]A^{1/2}\succ cI$ for
all $y$ with $\norm{F_{J,h}(y)}\le\rho_0\sqrt n$ and all diagonal
$0\le A\le(1+\eta)I$ near $V_y$
\ploc{(\S3, p.~7; (13.1); (17.2))}{accepted-inputs.tex:21--30; profile-positive.tex:16--30; cap-classical.tex:24--38}.
It gives a unique root $y_*$, $\norm{y-y_*}\le C\norm{F_{J,h}(y)}$
\ploc{(proof of Proposition~13.5)}{profile-positive.tex:1024--1033}, and
$\norm{\mu_hX-\tanh y_*}\le C$ \ploc{(3.2)}{accepted-inputs.tex:44--55}.
\end{itemize}

\paragraph{Role in the argument.}
Both routes need the mean, covariance and Dirichlet form of $\mu_s$, an SK
measure at field $h_s$. At a stable field, (3.2) (from
\sloc{OAI-AcceptedGap}{proof of Proposition~7.2}) puts the mean near
$\tanh y_*$, and Propositions~13.5 and~17.1 bound the covariance. Stability of
observation fields, and rarity of small-overlap approximate roots, are proved
by Kac--Rice under the planted law, where $(J,h_s)$ is jointly Gaussian given
$X$ (\S8--\S11; \ploc{Proposition~13.4}{profile-positive.tex:442--466};
\ploc{Lemma~17.3}{cap-classical.tex:298--314}), and moved to ordinary disorder
by $L$. Route~1 does this at every observation time $s$, with exceptional
rates in $k=ns^{3/2}$, down to the critical scale; Route~2 does it at its
boundary scale (\S17) to start the cap induction. The transfer recurs in
\ploc{the proof of Lemma~17.3}{cap-classical.tex:444--449} and, in reverse with
$\E L^2\le C(n+1)$, in \ploc{\S13.4}{profile-positive.tex:1704--1716}.

\paragraph{The earlier technique.}
EAMS sample from SK for $\beta<1/2$ by simulating the observation process of a
Gibbs sample, whose drift is the posterior mean, computed by AMP and natural
gradient descent on the TAP free energy; the analysis runs under a planted
model. In their bold notation, $\boldsymbol y$ is the observation (the
preprint's $h_s$); we write $\mathcal F_{\rm TAP}$ and $\mathrm H$ for their
TAP free energy and binary entropy.
\begin{itemize}
\item \emph{Model and observation.} $\boldsymbol A\sim\mathsf{GOE}(n)$ in the
normalization of $J$, and
$\mu_{\boldsymbol A,\boldsymbol y}(\boldsymbol x)\propto
\exp\{\frac\beta2\ip{\boldsymbol x}{\boldsymbol A\boldsymbol x}+\ip{\boldsymbol y}{\boldsymbol x}\}$
on $\{\pm1\}^n$ \sloc{ElAlaouiMontanariSellke2022-SamplingSKAlgorithmicSL}{(1.1), (2.1)}.
Given $\boldsymbol A$, the SDE
$d\boldsymbol y=\boldsymbol m(\boldsymbol y,t)\,dt+d\boldsymbol B$,
$\boldsymbol y(0)=0$, has the law of $t\boldsymbol x+\bar{\boldsymbol B}(t)$ with
$\boldsymbol x\sim\mu_{\boldsymbol A}$
\sloc{ElAlaouiMontanariSellke2022-SamplingSKAlgorithmicSL}{(1.4), p.~3; (4.12), pp.~11--12}.
\item \emph{TAP free energy with the observation as field}
\sloc{ElAlaouiMontanariSellke2022-SamplingSKAlgorithmicSL}{(2.3), p.~5; (4.53), p.~20}:
\begin{align*}
 \mathcal F_{\rm TAP}(\boldsymbol m;\boldsymbol y,q)&=-\tfrac\beta2\ip{\boldsymbol m}{\boldsymbol A\boldsymbol m}
 -\ip{\boldsymbol y}{\boldsymbol m}-\textstyle\sum_{i}\mathrm H(m_i)
 -\tfrac n4\beta^2(1-q)\bigl(1+q-2Q(\boldsymbol m)\bigr),\\
 \nabla\mathcal F_{\rm TAP}&=-\beta\boldsymbol A\boldsymbol m-\boldsymbol y
 +\operatorname{arctanh}(\boldsymbol m)+\beta^2(1-q)\boldsymbol m,
\end{align*}
with $Q(\boldsymbol m)=\frac1n\norm{\boldsymbol m}^2$ and $q=q_*(\beta,t)$ the
state-evolution fixed point. By the AMP analysis (Algorithm~1, p.~5;
Lemma~4.11) and local strong convexity (Lemma~4.12), Lemma~4.10 (p.~20) gives
a unique stationary point near the AMP output, close to the Gibbs mean and
Lipschitz in $\boldsymbol y$.
\item \emph{Planted model and likelihood}
\sloc{ElAlaouiMontanariSellke2022-SamplingSKAlgorithmicSL}{\S4.1, (4.1)--(4.9), pp.~10--11; Proposition~4.2; (4.14), p.~12}.
Under $\mu_{\rm pl}$, $\boldsymbol x$ is uniform,
$\boldsymbol A=\frac\beta n\boldsymbol x\boldsymbol x^{\mathsf T}+\boldsymbol W$

\subsection{S2. The spherical saddle-point method with a field (Kosterlitz--Thouless--Jones, Baik et al., Landon, Du--Huang), in one- and two-replica form}\label{f:S2}

\paragraph{Summary.}
The preprint's spherical reference model is built on the
Kosterlitz--Thouless--Jones (KTJ) saddle-point representation of the
spherical partition function with a field
\sloc{BaikEtAl2020-SphericalSpinGlassExternalField}{Lemma~3.1}, in the
Gaussian-conditioned-on-its-radius form evaluated at the mean radius
\sloc{DuHuang2026-CriticalSKFreeEnergyFluctuations}{proof of Prop.~4.4}.
A stronger form of its critical saddle location, with the same two-step
proof, is \sloc{Landon2020-CriticalSphericalSKFreeEnergy}{Prop.~3.1 and Lemma~3.4},
and its normalization bound is a weak form of Landon's Theorem~1.1. These are
\ploc{(5.4), (5.5) and Lemmas~5.3--5.4 in \S5.3}{profile-spectral.tex:364--656}.
Applied to two independent replicas, the same representation
Fourier-inverts the joint law of their squared lengths and overlap
\ploc{((7.2) and the paragraph after it, p.~27)}{profile-spectral.tex:1182--1213},
as in \sloc{DuHuang2026-CriticalSKFreeEnergyFluctuations}{Lemma~4.3; proof of Prop.~4.5};
the Laplace (double-contour) form is
\sloc{NguyenSosoe2018-OverlapCLTNearCriticalSphericalSK}{Lemma~4} and, with
a field, \sloc{BaikEtAl2020-SphericalSpinGlassExternalField}{Lemma~3.3}.
The method recurs in \S6--\S7, \S11--\S12, \S15 and \S19. The
preprint cites none of KTJ, Baik--Lee, Baik et al., Landon or
Nguyen--Sosoe, and cites the companion \cite{OAI-AcceptedCritical} only for
other content. It cites
\sloc{DuHuang2026-CriticalSKFreeEnergyFluctuations}{Thm.~1.6 and \S3.3}
\ploc{at the head of \S7 (p.~26)}{profile-spectral.tex:1133--1135} for the
zero-field partition-function comparison, but that section uses their \S4
only through its conclusion (their Prop.~3.1), and \S4, which contains the
representation, is not cited.
\textbf{Grade: \grade{Strong}}, because the saddle equation, the one- and
two-replica representations and the shell formula agree term by term after
renaming, and Landon proves a stronger saddle location by the same two
steps.

\paragraph{Setup.}
Shared notation is in Section~\ref{sec:overview}. For a field $h$, the
preprint's \emph{saddle} is the unique $\lambda>\lambda_{\max}(J)$ with
\[
 D=(\lambda-J)^{-1},\qquad m_{\mathrm o}=Dh,\qquad \Tr D+\norm{m_{\mathrm o}}^2=n,
\]
the preprint's \ploc{(5.4)}{profile-spectral.tex:367--377}; as a function of
$\lambda$ the left side decreases strictly from $\infty$ to $0$. The preprint states, without proof, that
the spherical Gibbs law
$\propto e^{x^{\mathsf T}Jx/2+h^{\mathsf T}x}\pi_{\mathrm o}(dx)$ is
precisely $N(m_{\mathrm o},D)$ conditioned on $\norm x^2=n$. A superscript
$0$ marks zero field, so $\Tr D^0=n$. The eigenvalues of $J$ are
$\lambda_1\ge\dots\ge\lambda_n$, with unit eigenvectors $v_k$; put
$\hat h_k=\ip{v_k}{h}$ and $D_k=(\lambda-\lambda_k)^{-1}$. Write $f_{m,C}(t)$ for the density
of $\norm Y^2$ at $t$ when $Y\sim N(m,C)$, and $f_C=f_{0,C}$. With
$\ell=\log\log(n+1000)$ and $d_0=n^{-1/3}\ell$,
\ploc{(5.5)}{profile-spectral.tex:379--386} derives $|\lambda^0-2|\le Cd_0^2$
and $\norm{D^0}=(\lambda^0-\lambda_{\max}(J))^{-1}\le Cnd_0$ from the edge
bounds \ploc{(5.1) of Lemma~5.1}{profile-spectral.tex:89--106} in two steps:
at $2+Cd_0^2$ the normalized trace deficit $1-\frac1n\Tr(2+Cd_0^2-J)^{-1}$ is of order
$d_0$, so $\lambda^0<2+Cd_0^2$; at distance $c/(nd_0)$ above
$\lambda_{\max}(J)$ the top pole alone closes that deficit.

For two replicas put $L=n^{2/3}$. Independent Gaussian replicas
$x_a=m_{\mathrm o}+D^{1/2}\eta_a$ ($a=1,2$, $\eta_a$ white) have the Gram
statistic $\mathsf G=\bigl((x_a^{\mathsf T}x_b-n\delta_{ab})/L\bigr)_{a,b}$,
the preprint's \ploc{(7.2)}{profile-spectral.tex:1184--1193}, and
characteristic function $\varphi(T)=\E\,e^{\mathrm i\Tr(T\mathsf G)}$,
$T\in\mathrm{Sym}_2(\R)$. Two independent spherical replicas are these
conditioned on $\mathsf G_{11}=\mathsf G_{22}=0$, so their overlap
$x_1^{\mathsf T}x_2/L$ has density
$f_{\mathsf G}(0,0,u)/(f_{\mathsf G_{11}}(0)f_{\mathsf G_{22}}(0))$ at $u$.

\paragraph{The earlier technique.}
\emph{The KTJ representation.} Fourier-inverting the constraint
$\norm\sigma^2=N$ on a standard Gaussian turns the spherical partition
function into one contour integral over the Lagrange multiplier $z$
\cite{NL-KosterlitzThoulessJones1976}.
\sloc{BaikLee2015-FluctuationsSphericalSKFreeEnergy}{Lemma~1.3, p.~5} state
it at zero field and
\sloc{BaikEtAl2020-SphericalSpinGlassExternalField}{Lemma~3.1, p.~11} with
a field. There $M$ is GOE with variances $(1+\delta_{ij})/N$, eigenvalues
$\lambda_1\ge\dots\ge\lambda_N$ and eigenvectors $u_i$;
$\mathcal H(\sigma)=-\frac12\sigma\cdot M\sigma-h\,g\cdot\sigma$ and
$Z_N=\int e^{-\beta\mathcal H}d\omega_N$, with $\omega_N$ normalized on the
sphere of radius $\sqrt N$ (eqs.~(1.1)--(1.4)). For every $\gamma>\lambda_1$,
\begin{gather*}
 Z_N=C_N\int_{\gamma-i\infty}^{\gamma+i\infty}e^{\frac N2G(z)}dz,\qquad
 C_N=\frac{\Gamma(N/2)}{2\pi i(N\beta/2)^{N/2-1}},\\
 G(z)=\beta z-\frac1N\sum_{i=1}^N\log(z-\lambda_i)+\frac{h^2\beta}N\sum_{i=1}^N\frac{n_i^2}{z-\lambda_i},
 \qquad n_i=u_i\cdot g .
\end{gather*}
By their eq.~(5.3) (p.~17), $G'$ increases from $-\infty$ to $\beta$ on
$(\lambda_1,\infty)$, so a unique saddle $\gamma>\lambda_1$ solves
$G'(\gamma)=0$.

\emph{The density-at-the-mean form.} Instead of steepest descent through
$\gamma$ (Baik--Lee, Landon), one can stop at the saddle: the Gibbs law is the Gaussian $N(0,(\gamma-M)^{-1})$ conditioned on its
squared radius, and $Z$ is a Gaussian normalization times that radius's
density at $N$, which at the saddle is its mean. Du and Huang use the
preprint's normalization, with their $W$ (not the planted noise) in place
of $J$. In
\sloc{DuHuang2026-CriticalSKOverlapDistribution}{\S1.2, eq.~(1.7), pp.~4--5},
$\gamma$ solves $\sum_i(\gamma-\lambda_i)^{-1}=N$ and $\mu^{\rm sph}_{N,1}$
is precisely the law of $\xi\sim N(0,(\gamma I-W)^{-1})$ conditioned on
$\norm\xi^2=N$. In the proof of
\sloc{DuHuang2026-CriticalSKFreeEnergyFluctuations}{Prop.~4.4, p.~22}, from
the KTJ formula (their Lemma~4.2, p.~18, with
$C_N=\Gamma(N/2)/(N/2)^{N/2-1}$), they set $a_k=\gamma-\lambda_k$ and
$\Xi_W=\sum_k\xi_k^2/a_k$ with $\xi_k$ i.i.d.\ $N(0,1)$, so
$\E\,\Xi_W=N$, and obtain the identity displayed below, where $\rho_W$ is
the density of $\Xi_W$.

\emph{Landon at criticality.}
\sloc{Landon2020-CriticalSphericalSKFreeEnergy}{eqs.~(2.3)--(2.5), pp.~4--5}
takes $H$ GOE with variances $1/N$ and $2/N$ and normalizes $Z_N(\beta)$ on
the sphere of radius $\sqrt N$, so $Z_N(1)$ is $Z_{\mathrm o}(0)$ with
$J=H$; the saddle solves $\beta=\frac1N\sum_i(\gamma-\lambda_i)^{-1}$
(eq.~(3.2), p.~9). At $\beta=1$, Prop.~3.1 (eq.~(3.3), p.~9) and Lemma~3.4
(p.~10) put both $N^{2/3}(\gamma-\lambda_1)$ and $N^{2/3}(\gamma-2)$ in
$[C_\varepsilon^{-1},C_\varepsilon]$ with probability at least
$1-\varepsilon$, in two steps. By eq.~(3.7), above the edge
$\frac1N\sum_j(E-\lambda_j)^{-1}$ falls below $1$ by at least
$cN^{-1/3}\sqrt{N^{2/3}(E-2)}-N^{-1/3}$, which bounds $\gamma$ from above;
the top-pole term $N^{-1/3}/(N^{2/3}(E-\lambda_1))$ of the decomposition
(3.5) (Lemma~3.2) closes this deficit near $\lambda_1$, which bounds
$\gamma-\lambda_1$ from below. Thm.~1.1 (eq.~(1.6),
p.~2) is the critical central limit theorem
$(\log Z_N(1)-\frac N4+\frac1{12}\log N)/\sqrt{\frac16\log N}\Rightarrow N(0,1)$,
where $f(1)=\frac14$ is the limiting free energy (eq.~(1.3), p.~1), computed
at zero field by KTJ
\sloc{BaikEtAl2020-SphericalSpinGlassExternalField}{p.~4}.

\emph{Two replicas.} In Laplace form, for spherical SK at zero field (any
$\beta>0$),
\sloc{NguyenSosoe2018-OverlapCLTNearCriticalSphericalSK}{Lemma~4, eqs.~(3.2)--(3.4), p.~6}
write $\langle e^{tR_{12}}\rangle$, normalized by the square of the
one-replica integral,
as a double integral over $z,w\in\gamma+\mathrm i\R$ with kernel
$(z-\lambda_i)(w-\lambda_i)-t^2/(4\beta^2N^2)$.
\sloc{BaikEtAl2020-SphericalSpinGlassExternalField}{Lemma~3.3, eqs.~(3.5)--(3.6), p.~12; App.~A, pp.~70--71}

and at $a=0$, where $\mathcal G_{\mathfrak R}$ has kernel
$(z-\lambda_i)(w-\lambda_i)-a^2$ and field term
$\frac{h^2\beta}N\sum_in_i^2(z+w-2\lambda_i+2a)/\bigl((z-\lambda_i)(w-\lambda_i)-a^2\bigr)$.

\subsection{S3. The Br\'ezin--Hikami duality for GOE characteristic polynomials with an external source}\label{f:S3}

\paragraph{Summary.}
The polynomial bound on the second moment of the Kac--Rice Jacobian
determinant \ploc{(Lemma~8.2, pp.~31--32)}{profile-roots.tex:95--192} rests on
an exact identity \ploc{(p.~32)}{profile-roots.tex:152--170} that writes
$\E\det(\,\cdot\,)^2$, for a GOE matrix with a diagonal shift and variance
profile, as a six-variable Gaussian average of a product of per-site factors.
It is the $k=2$ case of Br\'ezin and Hikami's dual
Grassmann representation of GOE characteristic polynomials with an external
source \sloc{BrezinHikami2001-CharPolyRealSymmetric}{\S6, eqs.~(88)--(89),
pp.~19--20, with (43), p.~10}: factoring out the variance profile turns it into
their diagonal source, and the preprint's derivation follows their \S3 step by
step. The preprint introduces the identity with ``The exact identity is'' and
derives it ``for completeness'', citing no source; the opening of \S8
\ploc{(p.~30)}{profile-roots.tex:10--15} credits only the Kac--Rice
organization and lists the ``polynomial determinant bound'' among results
``proved here''. Br\'ezin--Hikami appears nowhere in the preprint or its
companions. Grade: \grade{Strong}, because the identity and its derivation are
reused exactly and carry that bound.

\paragraph{Setup.}
For a candidate root $y$ of the TAP-type equation of Section~\ref{sec:overview}, put $b=1-q$,
$v=1-m^2\in(0,1]^n$ and $V=\operatorname{diag}v$
\ploc{(\S8, p.~30)}{profile-roots.tex:28--31}, and let $\mathsf Q_y$ be the
law of $J$ conditioned on $y$ being a root. The Kac--Rice formula
\ploc{(Lemma~8.1, p.~31)}{profile-roots.tex:46--65} integrates over $y$ an
explicit scalar root density times $\E_{\mathsf Q_y}[T\,\mathcal W]$, for any
nonnegative test $T(y,J)$, where
\[
 \mathcal W(y,J)=e^{-nb^2/2}\bigl|\det\bigl(I+(bI-J-2mm^{\mathsf T}/n)V\bigr)\bigr|
 \qquad\ploc{(8.1)}{profile-roots.tex:59--61}
\]
is the normalized absolute Jacobian. Lemma~8.2 states
$\E_{\mathsf Q_y}\mathcal W^2\le n^{C_D}$ on $\norm y\le n^D$. For an
unconditioned GOE matrix $W_0$, real $z$ and $x_i=1+z+bv_i$, the identity in
its proof \ploc{(p.~32)}{profile-roots.tex:152--162} is
\begin{equation*}
 \E_{W_0}\det\!\bigl((1+z)I+bV-\sqrt V W_0\sqrt V\bigr)^2
 =\E_H\prod_{i=1}^n\Bigl[(x_i+\mathrm i v_iH_1)(x_i+\mathrm i v_iH_2)
 +v_i^2\bigl(|h_1|^2+|h_2|^2\bigr)\Bigr],
 \tag{P}
\end{equation*}
where $H_1,H_2$ are real $N(0,1/n)$ and $h_1,h_2$ are circular complex
Gaussians with $\E|h_j|^2=1/n$, all independent. The bracketed terms are the
\emph{site factors}. The derivation \ploc{(p.~32)}{profile-roots.tex:162--170}
uses exterior (Grassmann) variables $\bar\psi_i^a,\psi_i^a$, $a=1,2$: averaging
$W_0$ adds $-[X_{11}^2+X_{22}^2+2X_{12}X_{21}+2PQ]/(2n)$ to the exponent, with
local symbols $X_{ab}=\sum_iv_i\bar\psi_i^a\psi_i^b$,
$P=\sum_iv_i\bar\psi_i^1\bar\psi_i^2$ and $Q=\sum_iv_i\psi_i^1\psi_i^2$, and
these terms are decoupled with $H_1,H_2,h_1,h_2$.

\paragraph{Role in the argument.}
On Route~1 (Section~\ref{sec:overview}), Sections~8--10 show that, under
the planted law, every root at the observed field passes scalar and matrix
diagnostics, which feed the \S11 comparison. Each diagnostic is first proved
without the weight $\mathcal W$, with failure probability $n^Ce^{-Bk}$, and
then transferred to the root count by
$\E_{\mathsf Q_y}[\one_F\mathcal W]\le\mathsf Q_y(F)^{1/2}
(\E_{\mathsf Q_y}\mathcal W^2)^{1/2}$. Polynomial accuracy is essential, since
$k=nr^3$ need only exceed $\log^{10}n$
\ploc{(\S8, p.~30)}{profile-roots.tex:45}: a loss of $e^{\varepsilon n}$, as in
the Kac--Rice works credited for the set-up
\cite{FanMeiMontanari2018-TAPFreeEnergyVariationalInference,CelentanoFanMei2021-LocalConvexityTAPZ2Sync},
would swamp $e^{-Bk}$. In the proof, symmetrization and a fixed-rank
comparison of $J$ under $\mathsf Q_y$ with $W_0$ reduce the bound to the left
side of (P), uniformly in real $|z|\le1/n$
\ploc{(pp.~31--32)}{profile-roots.tex:104--150}; then (P), a contour shift and
an elementary estimate give $\le Cn^{3/2}$
\ploc{(p.~32)}{profile-roots.tex:152--191}. The lemma is invoked at
\ploc{Proposition~8.3 (p.~33)}{profile-roots.tex:238--244},
\ploc{\S10.1 (p.~42)}{profile-orbits.tex:29--33} and, implicitly,
\ploc{Proposition~9.1 (p.~37)}{profile-matrices.tex:60--61}.

\paragraph{The earlier technique.}
Br\'ezin and Hikami \cite{BrezinHikami2001-CharPolyRealSymmetric} (arXiv
version; printed page numbers) compute
$F_k(\lambda_1,\dots,\lambda_k)=\E\prod_{\alpha=1}^k\det(\lambda_\alpha-\mathsf X)$
for a real symmetric $N\times N$ matrix $\mathsf X$ with weight
$e^{-\frac N2\operatorname{Tr}\mathsf X^2}$ ((21), p.~6), so
$\E\mathsf X_{ii}^2=1/N$ and $\E\mathsf X_{ij}^2=1/(2N)$, by trading the
$N\times N$ average for an integral over $k\times k$ matrices. Each determinant
is a Gaussian Grassmann integral,
$\det(\lambda-\mathsf X)=\int\prod_ad\bar c_a\,dc_a\,
e^{iN\bar c_a(\lambda\delta_{ab}-\mathsf X_{ab})c_b}$ ((6)--(7), p.~3), with
one copy $c_{a\alpha}$ per factor~$\alpha$. The GOE average is then Gaussian,
$\E e^{iN\operatorname{Tr}\mathsf XY}=e^{-\frac N4\operatorname{Tr}(Y^2+YY^{\mathsf T})}$
with $Y_{ab}=-\sum_\alpha\bar c_{a\alpha}c_{b\alpha}$ ((22), p.~6), and yields
two quartic terms: $\operatorname{Tr}Y^2=-\operatorname{tr}\gamma^2$ with
$\gamma_{\alpha\beta}=\sum_a\bar c_{a\alpha}c_{a\beta}$ ((10), (23)), already
present for the GUE, and the pairing term
$\operatorname{Tr}YY^{\mathsf T}=\operatorname{tr}\mathsf U\mathsf V$ with
$\mathsf U_{\alpha\beta}=\sum_a\bar c_{a\alpha}\bar c_{a\beta}$,
$\mathsf V_{\alpha\beta}=\sum_ac_{a\alpha}c_{a\beta}$ ((24)--(26), pp.~6--7),
which is specific to orthogonal symmetry. A Hubbard--Stratonovich transform
decouples the first with a $k\times k$ Hermitian $B$ ((11), p.~4) and the
second with a complex antisymmetric $D$ ((27), p.~7). The exponent is then a
sum over sites, so the Grassmann integral factorizes into one Pfaffian per
site ((28)--(30), p.~7):
\[
 F_k=\int dB\,dD\,e^{-N\operatorname{tr}(B^2+D^\dagger D)}\,[-\operatorname{Pf}\mathsf M]^N,
 \qquad
 \mathsf M=\begin{pmatrix}D&\Lambda-iB^{\mathsf T}\\-(\Lambda-iB)&D^\dagger\end{pmatrix}
 \quad\text{((31)--(32))},
\]
with $\Lambda=\operatorname{diag}(\lambda_\alpha)$ and $2k^2-k$ real
integration variables (pp.~7--8). For $k=2$, $D$ has a single entry $d$ and
\[
 -\operatorname{Pf}\mathsf M=|d|^2+(\lambda_1-iB_{11})(\lambda_2-iB_{22})+|B_{12}|^2
 \qquad\text{((42)--(43), pp.~9--10)},
\]
the quaternion determinant of a $2\times2$ self-dual quaternion matrix: the
GOE$\to$GSE duality (47). In \S6 (pp.~19--20) a diagonal source
$A=\operatorname{diag}(a_j)$ is added ((86), p.~19), and, following (88),
$F_k$ is read for $\E\prod_\alpha\det(\lambda_\alpha+A-\mathsf X)$ ((86) as
printed gives $\lambda_\alpha-A$; the sign of a diagonal source is immaterial
here). Being diagonal, the source
enters only the Pfaffian of site $j$, through $\Lambda\mapsto\Lambda+a_j\one$
in $\mathsf M^{(j)}$ ((88), p.~19), so
\[
 F_k=\int dB\,dD\,e^{-N\operatorname{tr}(B^2+D^\dagger D)}\prod_{j=1}^N[-\operatorname{Pf}\mathsf M^{(j)}]
 \qquad\text{((89), p.~20)}.
\]

\subsection{S4. Polchinski renormalization: trickle-down of Hessian bounds across caps}\label{f:S4}

\paragraph{Summary.}
The cap induction of the preprint carries a Hessian bound for the capped
potential $\phi_p$ from each spectral cap to the next finer one. Each cap step
is one discrete step of the Polchinski renormalization scheme of
Bauerschmidt--Bodineau \cite{BauerschmidtBodineau2019-SineGordonMultiscaleBakryEmery},
in the form set out in the survey of Bauerschmidt--Bodineau--Dagallier (BBD)
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{\S\S3.2, 3.4--3.5},
and the steps compose across caps by the semigroup property.
In stochastic-localization language it is the trickle-down covariance
induction of Anari--Koehler--Vuong (AKV)
\sloc{AnariKoehlerVuong2024-TrickleDownLocalizationSchemes}{\S3}, and its
scalar bound is their trickle-down of semi-log-concavity
\sloc{AnariKoehlerVuong2024-TrickleDownLocalizationSchemes}{Thm.~102}.
The step occupies \ploc{\S\S16.1--16.2}{cap-multiscale.tex:143--504}: the
convolution and transition \ploc{(16.3)--(16.4)}{cap-multiscale.tex:154--163},
the Hessian identity \ploc{(16.11)}{cap-multiscale.tex:388--395} and the
Brascamp--Lieb bound \ploc{(16.14)--(16.15)}{cap-multiscale.tex:454--504},
each matching a numbered equation of BBD with a sign flip ((16.14)--(16.15)
after recentring); it is
iterated in the \ploc{proof of Theorem~18.6}{cap-initialization.tex:1006--1012}
and reused in \ploc{\S18}{cap-initialization.tex:479--525}.
The preprint cites none of the sine-Gordon paper, BBD or
Bauerschmidt--Dagallier; it cites AKV and an earlier Bauerschmidt--Bodineau
paper only for their SK temperature thresholds, and near the identities it
credits only the Brascamp--Lieb inequality and, for the form of a two-sided
covariance bound, Chewi--Pooladian
\cite{ChewiPooladian2022-EntropicCaffarelliCovariance}
\ploc{(\S16.2, p.~129)}{cap-multiscale.tex:376--379}.
\textbf{Grade:} \grade{Strong}, because the whole one-step apparatus
(renormalized potential, fluctuation measure, Hessian identity,
Brascamp--Lieb bound under a margin) is reused intact and composed across the
caps by the semigroup property.

\paragraph{Setup.}

\begin{itemize}

capped Gibbs law $\mu_{p,h}\propto e^{x^{\mathsf T}J_px/2+h^{\mathsf T}x}$ is
the posterior of $X\sim\mu_J$ after a Gaussian observation of precision
$B_p$ with value $h$ \ploc{(\S2)}{model.tex:29--41}, so a larger $p$ means
more observation and a flatter interaction. The projected Hessian is
$K_p=\nabla^2_{H_p}(\phi_p|_{H_p})=\Cov_{\mu_{p,h}}(P_pX)$
\ploc{(\S16, p.~124)}{cap-multiscale.tex:17--23}.
\item \emph{Invariant.} With $R_p^H=(p^2+\alpha_p(h))^{-1}I_{H_p}$,

\[
 \norm{D_{\perp\perp}}\le\delta,\quad

 -K_0I_{H_p}\preceq D,\quad D_{ee}\le-0.30
\]
on the annulus $\mathcal A_p(\sigma)=\{h\in H_p:|\alpha_p(h)|\le\sigma p^2\}$
\ploc{(16.1)}{cap-multiscale.tex:8--12}. The negative entry $D_{ee}$ is the
radial deficit.

Then $J_c=J_p+\Delta$ and $\Delta_U=(1-x^2)p^2I_U$; inverses of $\Delta$ are

$\norm{\Delta_E^{1/2}L_p\Delta_E^{1/2}}<1$
\ploc{(\S16.1, p.~127)}{cap-multiscale.tex:264--269}.
\item \emph{Recentring.} With the child saddle parameter $\alpha$ frozen,

\end{itemize}

\paragraph{Role in the argument.}
This is Route~2 of Section~\ref{sec:overview}.
\ploc{Proposition~16.1}{cap-multiscale.tex:80--119} (``One cap step'')
propagates the invariant from $p$ to $c=\gamma p$, outside an event of
probability at most $Ce^{-ak_p}+Ce^{-a'k_c}$ over a fresh Haar refinement. The
step is iterated from the fixed initialization scale $p_0$ of \S\S17--18
down to $np^3\asymp\sqrt{\log n}$. The output,
\ploc{Theorem~18.6}{cap-initialization.tex:975--999}, is
$P_p\nabla^2\phi_p(h)P_p\preceq(1+\epsilon)p^{-2}P_p$ on a thin saddle annulus
(and $\preceq Cp^{-2}P_p$ on bounded field balls). These curvature bounds
control the information revealed along the forward observation path in
\ploc{\S19}{cap-entropy.tex:3--8, 427}, whose entropy-retention and layer
argument gives the log-Sobolev inequality
\ploc{(Theorem~19.3)}{cap-entropy.tex:856}. The same device recurs in \S18,
where the auxiliary potential is one convolution step and
\ploc{(18.13)}{cap-initialization.tex:518--525} is the Brascamp--Lieb bound
under the margin \ploc{(18.11)}{cap-initialization.tex:402--409}; in
single-step form in \ploc{\S19.4 (pp.~178--179)}{cap-entropy.tex:551--557}; and, closed by
a Poincar\'e estimate instead of Brascamp--Lieb, as the backward covariance
induction of \ploc{\S12.3}{profile-amplification.tex:633--645, 908--921} on
the isotropic route.

\paragraph{The earlier technique.}
Bauerschmidt--Bodineau's sine-Gordon paper
\cite{BauerschmidtBodineau2019-SineGordonMultiscaleBakryEmery} (not their
earlier JFA paper \cite{BauerschmidtBodineau2017-SimpleProofLSIHighTemp},
which the preprint cites and which makes a single such step) splits a
Gaussian covariance as $C_t=\int_0^t\dot C_s\,ds$ and integrates the
fluctuations out scale by scale. For a potential $V_0$, the renormalised
potential and the renormalisation semigroup are
\sloc{BauerschmidtBodineau2019-SineGordonMultiscaleBakryEmery}{(1.3)--(1.4), p.~2};
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{Def.~3.2, (3.11)--(3.12), p.~16}
\[
 e^{-V_t(\varphi)}=\mathbb E_{C_t}\bigl(e^{-V_0(\varphi+\zeta)}\bigr),\qquad
 \mathbf P_{s,t}F(\varphi)=e^{V_t(\varphi)}\,
 \mathbb E_{C_t-C_s}\bigl(e^{-V_s(\varphi+\zeta)}F(\varphi+\zeta)\bigr),
\]
and, by the semigroup property, from scale $s$ to $t$,
$V_t(\varphi)=-\log\mathbb E_{C_t-C_s}e^{-V_s(\varphi+\zeta)}$
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{proof of Lemma~3.11, p.~22}.
The kernel of $\mathbf P_{0,t}$ is the fluctuation measure
$\mu^\varphi_t(d\zeta)\propto e^{-V_0(\zeta)}\,\mathsf N(\varphi,C_t)(d\zeta)$
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{(3.17)--(3.18), p.~17}.
Differentiating gives $\nabla V_t(\varphi)=\E_{\mu_t^\varphi}[C_t^{-1}(\varphi-\zeta)]$
and
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{Lemma~3.12, (3.60)--(3.61), p.~23}
\[
 (f,\operatorname{Hess}V_t(\varphi)f)
 =\E_{\mu_t^\varphi}[(f,\operatorname{Hess}V_0(\zeta)f)]
   -\Var_{\mu_t^\varphi}\bigl((f,\nabla V_0(\zeta))\bigr)
 =(f,C_t^{-1}f)-\Var_{\mu_t^\varphi}\bigl((C_t^{-1}f,\zeta)\bigr).
\]
Bounding the variance by Brascamp--Lieb gives
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{Prop.~3.13, Proof~1, (3.63)--(3.65), p.~23}
\[
 \operatorname{Hess}V_t(\varphi)\ge
 \E_{\mu_t^\varphi}\bigl[\operatorname{Hess}V_0\,
   (C_t^{-1}+\operatorname{Hess}V_0)^{-1}C_t^{-1}\bigr],\qquad
 C_t^{1/2}\operatorname{Hess}V_t\,C_t^{1/2}\ge
 \E_{\mu_t^\varphi}\Bigl[\frac{\hat H_t}{\mathrm{id}+\hat H_t}\Bigr],
\]
with $\hat H_t=C_t^{1/2}\operatorname{Hess}V_0\,C_t^{1/2}$, in one step from
$0$ to $t$ for convex $V_0$.
For non-convex $V_0$ in one dimension the step needs a margin, $t<t_0$ with
$t_0^{-1}=-\inf V_0''$
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{Ex.~3.15, p.~26}.
Along the continuous flow the Hessian obeys
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{Lemma~3.11, (3.56)},
and Bauerschmidt--Bodineau's criterion also admits discrete decompositions
$C_0+C_1+\cdots$
\sloc{BauerschmidtBodineau2019-SineGordonMultiscaleBakryEmery}{Thm.~2.6}.
Bauerschmidt--Bodineau feed these bounds into a multiscale Bakry--\'Emery
criterion \sloc{BauerschmidtBodineau2019-SineGordonMultiscaleBakryEmery}{Thms.~1.2 and~2.5}.
For Ising models the fluctuation measure is an Ising measure with
spin covariance $\Sigma_t$, and
$\operatorname{Hess}V_t=C_t^{-1}-C_t^{-1}\Sigma_tC_t^{-1}$
\sloc{BauerschmidtDagallier2022-NearCriticalIsingLSI}{(1.23)}; BBD show that
both that criterion and Chen--Eldan's entropic stability then reduce to a
bound on $\Sigma_t$ uniform over external fields
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{Thm.~6.12, Ex.~6.15}.
Bauerschmidt--Dagallier use this to get a polynomial log-Sobolev inequality
for near-critical Ising models.

AKV run the same induction in stochastic-localization language, which BBD
identify with the Polchinski flow
\sloc{BauerschmidtBodineauDagallier2023-PolchinskiIntroduction}{\S4.5}. For a
linear-tilt localization $(\nu_t)$,
\sloc{AnariKoehlerVuong2024-TrickleDownLocalizationSchemes}{eq.~(8), p.~32}
\[
 \Cov(\nu_t)=\E[\Cov(\nu_{t+1})\mid\mathcal F_t]
 +\Cov(\nu_t)\Cov(Z_{t+1}\mid\mathcal F_t)\Cov(\nu_t),
\]
which carries covariance bounds backward from the more localized
$\nu_{t+1}$ to $\nu_t$; this is the ``trickle-down''. Their scalar
trickle-down of semi-log-concavity
\sloc{AnariKoehlerVuong2024-TrickleDownLocalizationSchemes}{Thm.~102, p.~76},
which they prove from its continuous-time form
\sloc{AnariKoehlerVuong2024-TrickleDownLocalizationSchemes}{(10)} rather than
by Brascamp--Lieb,
reads: if every tilt of $\mu$ has covariance $\preceq\gamma^{-1}I$ and
$T<\gamma$, then every tilt of $\mathcal G_{-T}\mu$, where
$d\mathcal G_{-T}\mu/d\mu\propto e^{T\norm x^2/2}$, has covariance
$\preceq(\gamma-T)^{-1}I$. Chen--Eldan's Lemma~50
\cite{ChenEldan2022-LocalizationSchemes} is the special case of a product
base measure on the cube, proved by one Gaussian decomposition step plus
Brascamp--Lieb and credited there to
\cite{BauerschmidtBodineau2017-SimpleProofLSIHighTemp}.

\paragraph{What is new in the preprint.}
\begin{itemize}
\item The iteration of the Brascamp--Lieb step over a discrete sequence of
caps. In the sources this bound is a single step (BBD
Prop.~3.13 from $0$ to $t$, BBD Ex.~3.15, AKV Thm.~102).
\item The radial deficit, regenerated at each step by completing the square,
Sherman--Morrison and Gaussian-bridge trace identities
\ploc{((16.16)--(16.20), pp.~130--131)}{cap-multiscale.tex:506--589}. The
sources have no radial or rank-one mechanism.
\item Conditional-Haar compression of transverse errors
\ploc{(\S16.4, pp.~131--136)}{cap-multiscale.tex:591--956}: sampled replicas,

rotation.
\item Bounds on typical fields only (annuli and bounded balls), where the
sources assume field-uniform bounds. The Brascamp--Lieb margin is restored by
the semiconcave extension (16.7), and matching lower bounds come from cutoff
Fisher (Cram\'er--Rao) tests

comparisons.

spectrally, which flattens the top of the spectrum.
\item The endpoint: entropy retention on stopped observation paths, then
layers (\S19), rather than the multiscale Bakry--\'Emery criterion.

\subsection{S5. The capacity--measure criterion for the log-Sobolev inequality}\label{f:S5}

\paragraph{Summary.}
Route~2 ends with \ploc{Lemma~19.1}{cap-entropy.tex:44--143} (p.~172), which
turns an energy estimate for a restricted class of bounded tests into the
log-Sobolev inequality. After a new first step that yields a lower bound on
capacity, its proof (\ploc{pp.~172--173}{cap-entropy.tex:94--143}) runs the
sufficiency half of the capacity--measure criterion for the log-Sobolev
inequality, stated by Barthe--Roberto
\sloc{BartheRoberto2003-SobolevInequalitiesRealLine}{\S4, remark~6)} and proved by
Barthe--Cattiaux--Roberto
\sloc{BartheEtAl2004-InterpolatedInequalitiesOrliczIsoperimetry}{Thms~11
and~13}, with the dyadic slices of Bakry--Coulhon--Ledoux--Saloff-Coste
\sloc{BakryEtAl1995-SobolevInequalitiesInDisguise}{\S10.1}. The same slicing,
without capacities, recurs in \ploc{the proof of
Theorem~14.2}{profile-entropy.tex:614--677} (pp.~111--112). The preprint cites
only Barthe--Roberto's Lemma~4, as a ``compare'' pointer for the final entropy
shift (\ploc{p.~173}{cap-entropy.tex:124--125}); the capacity, the criterion,
the slicing and the median split carry no citation, and the proof of
Theorem~14.2 cites nothing. Grade: \grade{Strong}, because the whole chain from
capacity to the log-Sobolev inequality is reused intact and only its last link
is credited.

\paragraph{Setup.}
For a probability measure $\eta$,
$\Ent_\eta(u)=\eta[u\log u]-\eta[u]\log\eta[u]$
\ploc{(\S19, p.~171)}{cap-entropy.tex:10--16}. With $\KL$ for the preprint's
$D(\cdot\Vert\cdot)$:
\begin{quote}
\textbf{Lemma 19.1} \ploc{(p.~172)}{cap-entropy.tex:44--64}. Let $\eta$ be a
strictly positive probability measure on a finite state space, $\mathcal E$ a
reversible Markov Dirichlet form, and $\min_x\eta(x)\ge e^{-M}$ with
$M\ge\log2$. Suppose that, for some $A>0$,
$\mathcal E(g)\ge A\,\eta[g^2]\log(1/\eta[g^2])$ for every $0\le g\le1$
supported on a set of measure at most $1/2$ with
$\KL\bigl(g^2\eta/\eta[g^2]\,\Vert\,\eta\bigr)\ge\frac12\log(1/\eta[g^2])$ and
$0<\log(1/\eta[g^2])\le M$. Then, with an absolute constant $C$,
\[
 \Ent_\eta(f^2)\le\frac{C(1+\log_+M)^2}{A}\,\mathcal E(f)
 \qquad\text{for every real-valued } f. \tag{19.3}
\]
\end{quote}
For a set $E$ with $\eta(E)=a\le1/2$, a test $h$ is \emph{admissible} if $h=1$
on $E$ and $h=0$ outside a set of measure at most $1/2$. The preprint defines
capacity as the infimum of $\mathcal E(h)$ over admissible $h$
(\ploc{p.~172}{cap-entropy.tex:91--92}); we write $\mathrm{cap}_\eta(E)$ for it.
Step~1 of the proof shows
\[
 \mathcal E(h)\ge\frac{cA}{(1+\log_+M)^2}\,a\log(1/a)
 \qquad\text{for every admissible } h. \tag{19.4}
\]
In the application $\eta=\mu_J$, $\mathcal E=\D_J$, $A=n^{-2/3-r}$, and the
hypothesis is the restricted estimate \ploc{(19.1)}{cap-entropy.tex:24--29} for
tests satisfying \ploc{(19.2)}{cap-entropy.tex:30--33}; on $\norm J\le3$ one
has $M=Cn$, and the factor $(\log n)^2$ is absorbed into the exponent
(\ploc{p.~173}{cap-entropy.tex:145--149}).

\paragraph{Role in the argument.}
Lemma~19.1 is the last link of route~2 (Section~\ref{sec:overview}).
\S\S19.2--19.6 prove (19.1) for tests satisfying (19.2) by following the
observation process to a product endpoint, which is designed for bounded tests
whose squared-density tilt has entropy comparable to $\log(1/\mu_J[f^2])$
(\ploc{p.~172}{cap-entropy.tex:34--37}). Lemma~19.1 turns this into
\ploc{Theorem~19.3}{cap-entropy.tex:856--874},
$\Ent_{\mu_J}(f^2)\le n^{2/3+\rho}\D_J(f)$, which gives the bound on
$C_{\mathrm{LS},J}$ in Theorem~1.1 (\ploc{\S20.3, p.~186}{dynamics.tex:196--199}).
The proof has three steps: (1)~equal-height layers give (19.4)
(\ploc{p.~172}{cap-entropy.tex:66--92}); (2)~dyadic layers give (19.3) for
$v\ge0$ supported on a set of measure at most $1/2$
(\ploc{pp.~172--173}{cap-entropy.tex:94--114}); (3)~a median split and an entropy
shift give (19.3) for all $f$ (\ploc{p.~173}{cap-entropy.tex:116--143}). Steps~2
and~3 use the hypothesis only through (19.4), and they are the reused part. The
same slicing, without capacities, removes the sup-norm restriction from the
support profile of route~1 in the proof of Theorem~14.2.

\paragraph{The earlier technique.}
Maz'ya \cite{NL-Mazya1985} reduced $L^2$ Sobolev-type inequalities to
inequalities between the capacity and the measure of sets, by cutting a
function into level slices. On a probability space this takes the following
form. (BR, BCR and BCLS abbreviate
\cite{BartheRoberto2003-SobolevInequalitiesRealLine},
\cite{BartheEtAl2004-InterpolatedInequalitiesOrliczIsoperimetry} and
\cite{BakryEtAl1995-SobolevInequalitiesInDisguise}; BCR page numbers refer to
arXiv:math/0407219v1. In the sources $A$ is a set; $\rho$ and $\varphi$ are
unrelated to the preprint's $\rho$ and $\phi_j$, and $p$, $q$ are BCLS's
exponents, unrelated to the preprint's $p$ and $p_j$.)
\begin{itemize}
\item \emph{Capacity of a set of measure at most $1/2$}
 \sloc{BartheEtAl2004-InterpolatedInequalitiesOrliczIsoperimetry}{\S5.2,
 pp.~30--31}:
 $\mathrm{Cap}_\mu(A)=\inf\{\int|\nabla f|^2d\mu:\ f|_A\ge1,\ \mu(f=0)\ge\frac12\}$.
 Barthe--Roberto \sloc{BartheRoberto2003-SobolevInequalitiesRealLine}{\S4,
 remark~6), p.~496} use tests $\one_A\le f\le\one_\Omega$ with $\mu(\Omega)=1/2$.
\item \emph{The criterion}
 \sloc{BartheRoberto2003-SobolevInequalitiesRealLine}{remark~6), pp.~495--496},
 stated without proof: the log-Sobolev constant is comparable, up to universal
 constants, to the best $\beta$ in
 $\beta\,\mathrm{Cap}_\mu(A)\ge\mu(A)\log(1/\mu(A))$ for $\mu(A)\le1/2$. The
 programme is to split $f$ at a median $m$ into
 $F_\pm=(f-m)\one_{\{f\gtrless m\}}$ and treat each half by Maz'ya's
 level-set argument.
\item \emph{Its proof}
 \sloc{BartheEtAl2004-InterpolatedInequalitiesOrliczIsoperimetry}{Thm~11,
 pp.~38--39; Thm~13, p.~45; Rem.~19, pp.~46--47}. Theorem~11: for $\rho>1$ and
 a non-decreasing $F\ge0$ with $F=0$ on $[0,2\rho]$, if
 $\mu(A)F(\rho/\mu(A))\le D\,\mathrm{Cap}_\mu(A)$ whenever
 $\mu(A)\le1/(2\rho)$, then
 \[
 \int f^2F\Bigl(\frac{f^2}{\int f^2d\mu}\Bigr)d\mu\le D\Bigl(\frac{\rho}{\sqrt\rho-1}\Bigr)^2\int|\nabla f|^2d\mu .
 \]
 Its proof sets $\Omega_k=\{f^2\ge2\rho^k\mu(f^2)\}$, bounds the left side by
 $\sum_{k\ge1}\mu(\Omega_k)\,2\rho^{k+1}\mu(f^2)F(2\rho^{k+1})$, and tests
 $\mathrm{Cap}_\mu(\Omega_k)$ with $h_k$, the function $|f|$ clipped between
 consecutive levels and rescaled to $[0,1]$. Theorem~13 with $\varphi=\log$,

\subsection{P1. Wang's signed two-spin curvature method}\label{f:P1}

\paragraph{Summary.}
The curvature inequality behind the terminal spectral gap of route~1 is
Wang's signed two-spin method
\sloc{Wang2026-OptimalMixingGlauberSKBetaHalf}{Lemma~3.4 and proof of
Thm~1.2, pp.~10--13}: condition on all but two spins $i,j$, expand the
two-spin Ising law in $U_{ij}=\tanh B_{ij}$ keeping the signed first-order
term, and sum over pairs, so that $\norm{\Lgen f}^2$ is at least the Dirichlet
form minus $a^{\mathsf T}Ua$, with $B$ the zero-diagonal interaction matrix and
$a_i=v_id_if$, up to higher-order terms in $U$. The
preprint imports it as Lemma~8.1 of AcceptedGap
\ploc{(\S3, p.~8)}{accepted-inputs.tex:91--94} and applies it in the proof
of \ploc{Proposition~13.6 (pp.~102--103)}{profile-positive.tex:1584--1698}.
At both places the preprint cites only the companion lemma, whose section
opens by crediting Wang's Lemma~3.4
\ploc{(AcceptedGap, \S8, p.~58)}{terminal.tex:8--11}; the preprint's one
citation of Wang is for his $\beta<1/2$ mixing theorem
\ploc{(\S1, p.~5)}{introduction.tex:182--185}.
Grade: \grade{Partial}, because Wang's two-spin estimate and its summation
over pairs are reused essentially intact, but their credit reaches the reader
only through the cited companion lemma.

\paragraph{Setup.}
AG denotes \cite{OAI-AcceptedGap}. Let
$\mu_h(x)\propto\exp(\frac12x^{\mathsf T}Bx+h^{\mathsf T}x)$ on
$\{-1,1\}^n$, with $B$ symmetric with zero diagonal, and let $P_i$ be the
heat-bath update at site $i$ (the $Q_i$ of Section~\ref{sec:overview}), an
orthogonal projection in $L^2(\mu_h)$. Put
\[
 D_i=I-P_i,\qquad H=\sum_iD_i=-\Lgen,\qquad
 d_if(x)=\tfrac12\bigl(f(x^{i\gets+1})-f(x^{i\gets-1})\bigr),
\]
let $m_i,v_i$ be the conditional mean and variance of $x_i$ given $x_{-i}$,
and $a_i=v_id_if$. Then $D_if=(x_i-m_i)d_if$ and
$\D_h(f)=\sum_i\mu_h[v_i(d_if)^2]=\sum_i\norm{D_if}^2_{L^2(\mu_h)}$.
Entrywise, $U=\tanh B$ (Wang's $T$), $Q=U\circ U$,
$r_3(U)=\max_i\sum_j|U_{ij}|^3$ and $r_4(U)=\max_i\sum_jU_{ij}^4$. In
Proposition~13.6, $B=\theta J^\circ$ with $0\le\theta\le1$, and $h$ is an
observation field at a large fixed time $T$ or later.

\paragraph{Role in the argument.}
Route~1 needs, at a large fixed observation time, a posterior gap that holds
uniformly over the law of the observed configuration $X$, and bounded
covariance and gap along a path that weakens the interaction to a product law.
\ploc{Proposition~13.6}{profile-positive.tex:1556--1582} supplies both,
outside probability $e^{-cn}$. Its proof restates the imported inequality
\ploc{(p.~102)}{profile-positive.tex:1585--1598}, checks its entrywise
hypothesis $\max|U_{ij}|\le u_0$ for every $B=\theta J^\circ$, bounds
$\mu_h[a^{\mathsf T}Ua+C|a|^{\mathsf T}Q|a|]$ by $\D_h(f)/4$ through a
large-field split (Section~\ref{f:P6}), and concludes
\ploc{(p.~103)}{profile-positive.tex:1683--1697} from
$\mu_h[(-\Lgen f)^2]\ge\D_h(f)/2$, by the spectral theorem, that every nonzero
eigenvalue of $-\Lgen$ is at least $1/2$, so $\Var_{\mu_h}f\le2\D_h(f)$ and
$\norm{\Sigma_h}\le2$. These
outputs feed the backward amplification
\ploc{(\S12.3, p.~71)}{profile-amplification.tex:428--435} and the proof of
Proposition~14.1 \ploc{(pp.~109, 111)}{profile-entropy.tex:427--429, 575--579}.

\paragraph{The earlier technique.}
Wang \cite{Wang2026-OptimalMixingGlauberSKBetaHalf} (arXiv v1, 23 August
2026; pages are the printed pages of v2) proves a deterministic Poincar\'e
criterion (Thm~1.2, p.~3) for Glauber dynamics of Ising models $\mu_{J,h}$ with
weak entries, uniformly in $h$: with $T=(\tanh J_{ij})$, the constant is
$\frac1n\bigl(1-\max\{\norm T,2\norm{T\circ T}\}-Cr_3(T)\bigr)$ whenever this
is positive. His generator is $\frac1n\sum_i(P_i-I)=-H/n$, with $D_i=I-P_i$.
\begin{itemize}
\item \emph{Curvature form} (Prop.~3.2, p.~9; Lemma~3.3, p.~10):
 $n^2\E_\mu\Gamma_2(f,f)=\norm{\sum_iD_if}^2=\sum_{i,j}\ip{D_if}{D_jf}$,
 $D_if=(x_i-m_i)d_if$ and $\norm{D_if}^2=\E_\mu[v_i(d_if)^2]$.
\item \emph{Two-spin estimate} (Lemma~3.4, p.~10; proof in \S3.2,
 pp.~11--13), which Wang presents as his new ingredient (\S1.1, p.~2). If
 $\mu(x)\propto\exp(J_{12}x_1x_2+h_1x_1+h_2x_2)$ and $t=\tanh J_{12}$ has
 $|t|\le\delta$, then for every $f$
 \[
 \ip{D_1f}{D_2f}\ge-t\ip{v_1d_1f}{v_2d_2f}
 -2t^2\ip{m_1v_1d_1f}{m_2v_2d_2f}-C|t|^3\bigl(\norm{D_1f}^2+\norm{D_2f}^2\bigr).
 \]
 The proof, with $a=\tanh h_1$, $b=\tanh h_2$, $A=1-a^2$, $B=1-b^2$ and
 $f=c+p(x_1-a)+q(x_2-b)+r(x_1-a)(x_2-b)$, writes the left side plus the two
 signed terms as $AB(\phi_0pq+\phi_1pr+\phi_2qr+\phi_{12}r^2)$ with
 $|\phi_0|\le C_0|t|^3$, $|\phi_1|+|\phi_2|\le C_0t^4$, $\phi_{12}\ge\frac12$;
 its steps are in the table below.
\item \emph{Summation} (proof of Thm~1.2, pp.~10--11): condition on
 $X_{-\{i,j\}}$, apply Lemma~3.4 to the conditional pair law, average, and sum
 over pairs. With $a=(v_id_if)_i$ and $b=(m_iv_id_if)_i$,
 \[
 n^2\E_\mu\Gamma_2(f,f)\ge\bigl(1-2Cr_3(T)\bigr)\sum_i\norm{D_if}^2
 -\E_\mu\bigl[a^{\mathsf T}Ta+2b^{\mathsf T}(T\circ T)b\bigr].
 \]
 Wang bounds the quadratic forms by
 $\max\{\norm T,2\norm{T\circ T}\}\sum_i\norm{D_if}^2$ and converts curvature

 he quotes from \sloc{NL-BakryGentilLedoux2014}{Prop.~4.8.3}. For SK,
 $\norm T\approx2\beta$, which gives a gap for $\beta<1/2$ (Thm~3.1, p.~8).

\paragraph{Setup.}
For a fixed spectrum and field, $\lambda>\lambda_{\max}(J)$ solves
\ploc{(5.4)}{profile-spectral.tex:369--373}
\[
 D=(\lambda-J)^{-1},\qquad m_{\mathrm o}=Dh,\qquad
 \Tr D+\norm{m_{\mathrm o}}^2=n,
\]
and the spherical Gibbs law is $N(m_{\mathrm o},D)$ conditioned on
$\norm x^2=n$ \ploc{(\S5.3, p.~17)}{profile-spectral.tex:375--378}. When
$\lambda-2\asymp d^2$ with $d_0\le d\lesssim1$, one has $\Tr D^2\asymp n/d$
and $\norm D/(\Tr D^2)^{1/2}=O((nd^3)^{-1/2})=o(1)$
\ploc{(pp.~17--18)}{profile-spectral.tex:430--435, 471--474}.
\begin{itemize}
\item \emph{Lemma~5.2 (Density at the mean)}
 \ploc{(p.~17)}{profile-spectral.tex:392--398}. Let $g$ be a
 finite-dimensional standard Gaussian vector and
 $Q=g^{\mathsf T}Ag+b^{\mathsf T}g$ with $A\succeq0$ and $\sigma^2=\Var Q>0$.
 The density of $Q$ at $\E Q$ is at least $c/\sigma$ for an absolute $c>0$,
 and, with a smaller constant, at a displacement $o(\sigma)$ along any
 sequence of such forms. The proof normalizes to
 $\sum_ia_i(g_i^2-1)+2\sum_ib_ig_i$, $a_i\ge0$,
 $2\sum_ia_i^2+4\sum_ib_i^2=1$; Fourier inversion handles the case in which
 neither a single $a_i$ nor the linear part dominates.
\item \emph{Shell conditioning} \ploc{(p.~18)}{profile-spectral.tex:441--458}.
 Put $x=m_{\mathrm o}+D^{1/2}g$, $S=\norm x^2$, $Z=V^{\mathsf T}g$ for a
 matrix $V$ with a fixed number of orthonormal columns, and
 $P=I-VV^{\mathsf T}$. By (5.4), $\E S=n$. Then
 $p_{Z\mid S=n}(z)=p_Z(z)\,f_{S\mid Z=z}(n)/f_S(n)$. Given $Z=z$, $x$ is
 Gaussian with mean $m_{\mathrm o}+D^{1/2}Vz$ (the ``translated mean'') and
 covariance $D^{1/2}PD^{1/2}$, whose nonzero eigenvalues are those of the
 compression $PDP|_{\ran P}$.
\end{itemize}

\paragraph{The earlier technique.}

\begin{itemize}
\item \emph{Lemma~3} (p.~5). If $\alpha_1^2+\dots+\alpha_n^2=1$ and

 \[

 \]
 Proof (pp.~5--6): $-\log|f(t)|=\frac14\sum_k\log(1+4\alpha_k^2t^2)$ is concave in

 the modulus: $|\E e^{it(Z-a)^2}|=(1+4t^2)^{-1/4}\exp\{-2a^2t^2/(1+4t^2)\}$

 $M(W_a)\le2/\sqrt{A_1+B_1}\le2/\sqrt{A_1}$.
\end{itemize}

\subsection{P6. Large-field reduction to sparse principal submatrices (Bandeira--El Alaoui--R\"odder)}\label{f:P6}

\paragraph{Summary.}
The preprint reuses a mechanism of Bandeira, El Alaoui and R\"odder (BER
below) \sloc{BandeiraElAlaouiRodder2026-GlauberHighOverlapGibbs}{Thm.~1.1 and
proof of Lemma~2.3}: a large external field leaves only a small,
configuration-dependent set of weakly biased coordinates, and a norm bound
holding uniformly over all sparse principal submatrices makes the Gibbs
measure effectively high-temperature on that set. It is used twice, each time
to carry a near-critical posterior to a product law at a large observation
time: in the stability verification of
\ploc{\S19.4 (p.~178)}{cap-entropy.tex:530--549}, and in the proof of
\ploc{Proposition~13.6 (pp.~102--103)}{profile-positive.tex:1556--1698}. BER
is cited nowhere: \S19.4 cites nothing, and \S13.4 cites the companion's net
lemma, whose lead-in paragraph
\ploc{(companion \S8, p.~61)}{terminal.tex:217--223} credits BER.
\textbf{Grade: \grade{Partial}}, because only BER's two elementary steps (the weak-field
count and the sparse-submatrix bound) are reused, and the decisive block split
is the companion's.

\paragraph{Setup.}
AG denotes the companion \cite{OAI-AcceptedGap}. In \S\S17--19, $\beta<1$ is
not an inverse temperature but a fixed interaction coefficient close to~$1$
\ploc{(\S18, pp.~158--159)}{cap-initialization.tex:39--66}, and
$|u|_n=n^{-1/2}\norm u$.
\begin{itemize}
\item \emph{Stability test.} For an interaction $J_*$ with entries of variance
 $j/n$, \ploc{(17.1)}{cap-classical.tex:17--21} sets $m=\tanh y$,

 $F_h(y)=y+(jrI-J_*)m-h$, whose roots are TAP solutions. A field $h$ passes

  I+A^{1/2}\bigl(jrI-J_*-\tfrac{2j}{n}mm^{\mathsf T}\bigr)A^{1/2}\succeq bI .
 \]
 At $A=V$ this is the symmetrized TAP Jacobian. For such fields with

\paragraph{The earlier technique.}
BER prove polynomial-time Glauber mixing for quadratic Gibbs measures, SK at
every $\beta$ included, once the external field is large
\sloc{BandeiraElAlaouiRodder2026-GlauberHighOverlapGibbs}{\S1.1, Thm.~1.1 and Cor.~1.2, p.~3}.
We write $K$ for their interaction matrix; their $m$ is a dimension.
\emph{Theorem~1.1}: if
$\max_{|\mathcal S|\le\delta n}\norm{K_{\mathcal S\mathcal S}}\le\frac15$
for some $\delta\in(0,1)$, then for some $\theta$ independent of $n$, lazy
Glauber dynamics for
$\nu(x)\propto\exp(\frac12x^{\mathsf T}Kx+\theta\one^{\mathsf T}x)$ mixes in
polynomial time. The proof combines two observations.
\begin{itemize}
\item \emph{Few weak coordinates.} A field $\theta\one+v$ with
 $\norm v=O(\sqrt n)$ ($v$ comes from pinning, Proposition~3.2, p.~10) is
 below $\theta/2$ only on a small set, by a Chebyshev count (proof of
 Lemma~2.3, p.~7). A spin-flip injection then shows that most spins align
 with the field (Lemma~2.3, Corollary~2.4).
\item \emph{High temperature on small sets.} The remaining randomness lives
 on a small set: the free spins after pinning (\S3.2, p.~11), or the set
 $\mathcal D$ where two replicas disagree, on which the halved replica
 difference is Gibbs with interaction $2K_{\mathcal D\mathcal D}$
 (Lemma~2.1, p.~5). The hypothesis then bounds its covariance (eq.~(11)).
\end{itemize}
The GOE matrix satisfies the hypothesis by Gaussian concentration and a union
bound over supports, which BER call routine (\S3.3, pp.~11--12). AG adapt the
mechanism at fixed $j<1$, crediting BER's Theorem~1.1 just before its
Lemma~8.2 \ploc{(\S8, p.~61)}{terminal.tex:217--223}. AG's
\ploc{Lemma~8.2}{terminal.tex:225--327} proves the sparse bounds with $9^k$
nets, and the proof of its \ploc{Proposition~8.3}{terminal.tex:334--433}
counts weak-field sites, then splits the curvature form into a block on the
weak set $S(x)$, controlled by the sparse bound, and cross and complementary
blocks, controlled by the global norm times $2\sqrt\eta$ and $\eta$,
respectively, where $\eta=\operatorname{sech}^2(T/4)$.

\subsection{C1. The spectral-cap route is not needed for the log-Sobolev exponent}\label{f:C1}

\paragraph{Summary.}
Route~1 ends with the support profile of
\ploc{Theorem~14.2}{profile-entropy.tex:599--612} (p.~111). By the
Hermon--Peres characterization of the log-Sobolev constant through Dirichlet
eigenvalues of sets
\sloc{HermonPeres2016-L2MixingHypercontractivityHitting}{Thm~1.2, p.~3 of
arXiv v3}, this profile alone gives
$C_{\mathrm{LS},J}<17c_\omega^{-1}n^{2/3+\omega}$; Diaconis--Saloff-Coste
\sloc{DiaconisSaloffCoste1996-LogSobolevFiniteMarkovChains}{Cor.~3.11,
p.~728}, applied to the $L^2$ bound in the proof of
\ploc{Lemma~20.1}{dynamics.tex:83--93} (p.~184), gives the same exponent up to a
factor $\log n$. The preprint instead takes the upper bound on
$C_{\mathrm{LS},J}$ in Theorem~1.1 only from
\ploc{Theorem~19.3}{cap-entropy.tex:856--874}, the endpoint of the
spectral-cap route \ploc{\S\S15--19 (pp.~112--183)}{cap-static.tex:1 to
cap-entropy.tex:874}, and states in \ploc{\S2 (p.~7)}{model.tex:70--72} that
this inequality ``will require its own entropy and layer argument''. It cites
Diaconis--Saloff-Coste only for Theorems~3.5 and~3.7, and does not cite
Hermon--Peres. Conversely, Goel--Montenegro--Tetali
\sloc{GoelEtAl2005-MixingSpectralProfile}{Lemma~4.2 and the remark after it,
p.~15 of arXiv v1}, also cited, turn Theorem~19.3 back into the support
profile, so the two routes end at statements equivalent up to constant
factors. Grade:
\grade{Major}, because a component of the main theorem becomes a citation and
about 70 pages become unnecessary for Theorem~1.1.

\paragraph{Setup.}
Shared notation is in Section~\ref{sec:overview}. Write
$h_t^x=e^{t\Lgen_J}(x,\cdot)/\mu_J$ for the density at per-site time $t$ from
the start $x$, and
\[
 \tau_2=\inf\bigl\{t>0:\ \max_x\norm{h_t^x-1}_{L^2(\mu_J)}\le1/e\bigr\}
\]
for the worst-start $L^2$ mixing time of $e^{t\Lgen_J}$. Lemma~20.1 assumes
\ploc{(20.1)}{dynamics.tex:17--22}: $\D(f)\ge\alpha\,\mu(f^2)\log(1/p)$ for
every $f\ge0$ with $0<p=\mu(f>0)\le\frac12$. On the event of Theorem~14.2 this
holds with $\mu=\mu_J$ and $\alpha=c_\omega n^{-2/3-\omega}$. Its proof
follows $V_t=\mu_J\bigl((h_t^x)^2\bigr)=1+\norm{h_t^x-1}_2^2$ and uses
$M=\max\{8,\mu_{J,\min}^{-1}\}$, where $\mu_{J,\min}\ge e^{-(3+\log2)n}$ on
$\norm J\le3$ \ploc{(20.5)}{dynamics.tex:144--146}. The sources'
constants are written $\alpha_{\mathrm{HP}}$, $\alpha_{\mathrm{DSC}}$ and
$\rho_{\mathrm{GMT}}$, distinct from the preprint's $\alpha$ and the slack
$\rho$ of Theorem~19.3.

\paragraph{Role in the argument.}
The mixing upper bounds come from Lemma~20.1 with Theorem~14.2
\ploc{(p.~185)}{dynamics.tex:137--148}. In
\ploc{\S20.3 (p.~186)}{dynamics.tex:196--201} the median consequence of
Theorem~14.2 gives the upper bound on $R_J$, and Theorem~19.3 alone is named
for the upper bound on $C_{\mathrm{LS},J}$. Route~2 exists to prove Theorem~19.3. It comprises the
cube--sphere comparison on caps (\S15), the radial-deficit induction (\S16),
its initialization (\S\S17--18), and entropy retention closed by Lemma~19.1
(\S19). Elsewhere Theorem~19.3 serves only as a second proof of the mixing
bounds in \ploc{\S20.2 (p.~185)}{dynamics.tex:150--185} and, through
$C_{\mathrm{EP},J}\le C_{\mathrm{LS},J}/4$, in
\ploc{Corollary~20.2}{dynamics.tex:249--263}. Route~1 does not use route~2:
\S\S5--14 refer to no result of \S\S15--19, and \S20.1 uses only Theorem~14.2
and (20.5).

\paragraph{The earlier results.}
\begin{itemize}
\item \emph{Hermon--Peres.} For a finite irreducible chain $(\Omega,P,\pi)$
 with $Q=(P+P^*)/2$, set $\mathcal E(f)=\ip{(I-Q)f}{f}_\pi$,
 $c_{\mathrm{LS}}=\inf\{\mathcal E(f)/\Ent_\pi(f^2):f\text{ non-constant}\}$
 and $t_{\mathrm{LS}}=1/c_{\mathrm{LS}}$ ((3.11), p.~9). For $A\subsetneq\Omega$,
 $\lambda(A)$ is the smallest eigenvalue of $I-Q_A$, with $Q_A$ the
 restriction of $Q$ to $A$ (p.~3); by the Rayleigh quotient,
 $\lambda(A)=\min\{\mathcal E(f)/\pi(f^2):f\ne0,\ \operatorname{supp}f\subset A\}$.
 $\mathrm{Con}_\delta$ is the set of connected $A$ with $\pi(A)\le\delta$
 (p.~2). With
 \[
  \alpha_{\mathrm{HP}}=\min_{A\in\mathrm{Con}_{1/2}}\frac{\lambda(A)}{|\log\pi(A)|},\qquad
  \kappa=1/\alpha_{\mathrm{HP}} \qquad\text{(HP (1.5))},
 \]
 Theorem~1.2 (p.~3) states, for every irreducible chain on a finite state
 space,
 \[
  \kappa\le t_{\mathrm{LS}}\le2(\kappa+t_{\mathrm{rel}}(1+\log49))
  \le2\Bigl(1+\frac{1+\log49}{\log2}\Bigr)\kappa<17\kappa
  \qquad\text{(1.7)},
 \]
 where $t_{\mathrm{rel}}$ is the inverse of the smallest nonzero eigenvalue of
 $I-Q$, and the third step uses $t_{\mathrm{rel}}\log2\le\kappa$ (1.6).
\item \emph{Diaconis--Saloff-Coste.} For an irreducible Markov kernel $K$,
 $H_t=e^{-t(I-K)}$ (p.~695), $h_t^x=H_t^x/\pi$,
 $\mathcal E(f,f)=\frac12\sum_{x,y}|f(x)-f(y)|^2K(x,y)\pi(x)$ and
 $\alpha_{\mathrm{DSC}}=\min\mathcal E(f,f)/\Ent_\pi(f^2)$ ((1.4)--(1.7),
 p.~696; DSC write the entropy as $\sum|f|^2\log(|f|^2/\norm f_2^2)\pi$).
 Corollary~3.11 (p.~728): if $(K,\pi)$ is reversible and
 $\tau=\inf\{t>0:\sup_x\norm{h_t^x-1}_2\le1/e\}$, then
 $\frac1{2\alpha_{\mathrm{DSC}}}\le\tau\le\frac1{4\alpha_{\mathrm{DSC}}}
 \bigl(4+\log\log\frac1{\pi_*}\bigr)$.
\item \emph{Goel--Montenegro--Tetali.} For the same $\mathcal E$,
 $\rho_{\mathrm{GMT}}=\inf\mathcal E(f,f)/\Ent_\pi(f^2)$ (p.~15) and
 $\lambda_0(A)=\inf_{f\in c_0^+(A)}\mathcal E(f,f)/\pi(f^2)$, over nonnegative
 $f$ supported in $A$ (defined p.~4; variational form p.~9). Lemma~4.2 gives
 $\Lambda(r)\ge\rho_{\mathrm{GMT}}\log(1/r)/(1-r)$ for the spectral profile
 $\Lambda$, and the remark after it gives
 $\lambda_0(A)\ge\rho_{\mathrm{GMT}}\log(1/\pi(A))$ (p.~15).
\end{itemize}

is irreducible because every single-spin flip has positive heat-bath
probability. The objects then match as follows.

\begin{enumerate}
\item \emph{Hermon--Peres.} If $\mu_J(A)\le\frac12$ and $f\ne0$ is supported in
 $A$, then $p\le\mu_J(A)$, so (14.16) gives
 $\D_J(f)/n\ge c_\omega n^{-5/3-\omega}\log(1/\mu_J(A))\,\mu_J(f^2)$. Hence
 $\lambda(A)\ge c_\omega n^{-5/3-\omega}|\log\mu_J(A)|$ for every such $A$,
 connected or not, and $\alpha_{\mathrm{HP}}\ge c_\omega n^{-5/3-\omega}$. Then
 (1.7) gives $nC_{\mathrm{LS},J}<17\kappa\le17c_\omega^{-1}n^{5/3+\omega}$ on the
 event of Theorem~14.2, where (14.16) holds for all $f$ at once. With
 $\omega=\delta/2$ this is $C_{\mathrm{LS},J}\le n^{2/3+\delta}$ with probability
 tending to one, for every $\delta>0$.
\item \emph{Diaconis--Saloff-Coste.} Corollary~3.11 reads
 $nC_{\mathrm{LS},J}/2\le n\tau_2$, that is $C_{\mathrm{LS},J}\le2\tau_2$. The
 proof of Lemma~20.1 shows, for every start, $V_t\le8$ once
 $t\ge t_0=\alpha^{-1}\log\frac{\log(M/4)}{\log2}$ and
 $V_{t_0+t}-1\le7e^{-2\alpha(\log2)t}$
 \ploc{(p.~184)}{dynamics.tex:86--93}. So
 $\tau_2\le t_0+\frac{2+\log7}{2\alpha\log2}$, and (20.5) gives
 $C_{\mathrm{LS},J}\le2\alpha^{-1}(\log(6n)+3)=O(n^{2/3+\omega}\log n)$.
\item \emph{Converse.} With $\rho_{\mathrm{GMT}}=1/(nC_{\mathrm{LS},J})$, the remark after
 GMT Lemma~4.2, applied to $|f|$ with $A=\{f\ne0\}$, gives
 $\D_J(f)\ge C_{\mathrm{LS},J}^{-1}\log(1/p)\,\mu_J(f^2)$. So Theorem~19.3,
 $C_{\mathrm{LS},J}\le n^{2/3+\rho}$, yields (14.16) with $n^{-2/3-\rho}$ in
 place of $c_\omega n^{-2/3-\omega}$.
\end{enumerate}

\paragraph{What the cap route still contributes.}
\begin{itemize}
\item A second proof of all the upper bounds in Theorem~1.1, through
 hypercontractivity (\S20.2), that does not pass through Theorem~14.2.
\item Results on capped Gibbs measures that nothing outside \S\S15--19 uses:

 $p_{\min}(n)\le p\le p_{\mathrm{top}}$

 above loses only the factor~17.
