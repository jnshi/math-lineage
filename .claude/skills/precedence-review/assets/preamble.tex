% Findings are numbered by LaTeX, so nobody types an ID.  In the main file,
% \findinggroup{S} (or {P}, {C}) after each findings \section sets the ID prefix
% and restarts the count; IDs follow the order of the \input lines.  Each
% fragment opens with
%   \finding{<title>}{<slug>}   or   \finding[<short title>]{<title>}{<slug>}
% which prints \subsection{S3. <title>} (with "S3. <short title>" in the table
% of contents and PDF bookmarks when a short title is given) and does
% \label{f:<slug>}, so \ref{f:<slug>} prints the ID "S3" as a link.  Refer to
% another finding as finding~\ref{f:<slug>}.  Before any \findinggroup the
% IDs print as ?1, ?2, ... (as in a standalone test of one fragment).
\makeatletter
\newcounter{finding}
\newcommand{\finding@prefix}{?}
\renewcommand{\thefinding}{\finding@prefix\arabic{finding}}
\newcommand{\findinggroup}[1]{\gdef\finding@prefix{#1}\setcounter{finding}{0}}
\newcommand{\finding}[3][]{%
  \stepcounter{finding}%
  \protected@edef\finding@id{\thefinding}% fix the ID now: "S3", not a counter reference
  \edef\finding@next{\noexpand\finding@sect{\finding@id}}%
  \finding@next{#1}{#2}{#3}}
% #1 ID (already expanded), #2 short title (may be empty), #3 title, #4 slug.
\newcommand{\finding@sect}[4]{%
  \if\relax\detokenize{#2}\relax
    \subsection{#1. #3}%
  \else
    \subsection[#1. #2]{#1. #3}%
  \fi
  \def\@currentlabel{#1}% \ref prints the ID; hyperref links it to this subsection
  \label{f:#4}}
\makeatother
